--Start of the Game Boy video RAM, holding the sprite and background tile patterns. Copy data here with memcpyonhblank or while the LCD is turned off, never directly.
c; VRAM ; GAMEBOY; a; $8000
--Tile pattern area used by the background and window layers, and where a character set is normally placed.<br>
--<code>Screens::CopyBankHB(#font, VRAM_CHARSET, 4); // 4x$100 bytes, safely on hblank</code>
c; VRAM_CHARSET ; GAMEBOY ; a; $9000
--First of the two 32x32 background tile maps. Each byte selects the tile drawn in that cell.<br>
--<code>Screens::ClearScreen(SCREEN0, $80);</code>
c; SCREEN0 ; GAMEBOY; a; $9800
--Second 32x32 background tile map, typically used for the window layer or for double buffering.
c; SCREEN1 ; GAMEBOY; a; $9C00
--Background palette register. It packs the four shades used by background and window tiles, two bits per colour.<br>
--<code>BACKGROUND_PALETTE:=%10010011; // 11 is black, 00 is white</code>
c; BACKGROUND_PALETTE; GAMEBOY; a; $FF47
--First sprite palette. Colour 0 is always transparent for sprites.<br>
--<code>OBJECT0_PALETTE:=%10010011;</code>
c; OBJECT0_PALETTE; GAMEBOY; a;$FF48
--Second sprite palette, selected per sprite through the palette bit in the sprite attribute byte.
c; OBJECT1_PALETTE; GAMEBOY ; a;$FF49
--ROM bank select. Writing a bank number here swaps that bank into $4000 on cartridges with a memory bank controller.<br>
--<code>CURRENT_BANK:=charBank; // point at the bank holding the graphics</code>
c; CURRENT_BANK ; GAMEBOY; a; $2000
--Bit mask for the sprite attribute byte that mirrors the sprite horizontally.<br>
--<code>flags:=SPRITE_FLIP_X;</code>
c; SPRITE_FLIP_X; GAMEBOY; b; %00100000
--Bit mask for the sprite attribute byte that mirrors the sprite vertically. Combine with SPRITE_FLIP_X to rotate 180 degrees.<br>
--<code>flags:=SPRITE_FLIP_X or SPRITE_FLIP_Y;</code>
c; SPRITE_FLIP_Y; GAMEBOY; b; %01000000
--Vertical hardware scroll of the background layer, 0 to 255. Wraps around the 256 pixel tile map.<br>
--<code>SCROLLY:=SCROLLY+1; // scroll the background upwards</code>
c; SCROLLY ; GAMEBOY; a ; $FF42
--Horizontal hardware scroll of the background layer, 0 to 255.<br>
--<code>SCROLLX:=SCROLLX+1;</code>
c; SCROLLX ; GAMEBOY; a ; $FF43
--Start of High RAM, the small block of fast RAM that stays readable while an OAM sprite transfer is running.
c; DMA; GAMEBOY; a; $FF80
--LCD status register. Holds the current drawing mode and the interrupt enable bits for hblank, vblank and the LYC compare line.
c; LCDCSTAT; GAMEBOY; a; $FF41
--Top line of the window layer. Set it to 144 or above to hide the window.<br>
--<code>WINDOWY:=112; // window covers the bottom of the screen</code>
c; WINDOWY; GAMEBOY; a; $FF4A
--Left edge of the window layer, offset by 7. A value of 7 puts the window at the far left.<br>
--<code>WINDOWX:=7;</code>
c; WINDOWX; GAMEBOY; a; $FF4B
--Bit mask for the right direction in the byte filled in by joypad.<br>
--<code>if (pad1&PAD_RIGHT=PAD_RIGHT) then x:=x+1;</code>
c; PAD_RIGHT; GAMEBOY; b; 16
--Bit mask for the left direction in the byte filled in by joypad.
c; PAD_LEFT; GAMEBOY; b; 32
--Bit mask for the down direction in the byte filled in by joypad.
c; PAD_DOWN; GAMEBOY; b; 64
--Bit mask for the up direction in the byte filled in by joypad.
c; PAD_UP; GAMEBOY; b; 128
--Bit mask for the A button in the byte filled in by joypad.<br>
--<code>if (pad1&PAD_BUTTON_A=PAD_BUTTON_A) then Jump();</code>
c; PAD_BUTTON_A; GAMEBOY; b; 1
--Bit mask for the B button in the byte filled in by joypad.
c; PAD_BUTTON_B; GAMEBOY; b; 2
--Bit mask for the START button in the byte filled in by joypad.
c; PAD_BUTTON_START; GAMEBOY; b; 8
--Bit mask for the SELECT button in the byte filled in by joypad.
c; PAD_BUTTON_SELECT; GAMEBOY; b; 4

# NES-specific

--Main PPU control register: selects the name table, the address increment, the sprite and background pattern tables and enables the vblank interrupt.<br>
--<code>PPUCTRL:=%10010000; // nmi on, background uses pattern table 1</code>
c; PPUCTRL; NES; a; $2000
--PPU mask register: switches the background and sprites on or off, and controls the leftmost column and colour emphasis.<br>
--<code>PPUMASK:=%00011110; // show background and sprites</code>
c; PPUMASK; NES; a; $2001
--PPU status register. Reading it returns the vblank and sprite zero hit flags, and resets the address latch used by PPUADDR and PPUSCROLL.
c; PPUSTATUS; NES; a; $2002
--Selects which byte of the sprite attribute memory the next OAMDATA write goes to. Normally set to 0 before a transfer.
c; OAMADDR; NES; a; $2003
--Reads or writes one byte of sprite attribute memory at the current OAMADDR, stepping the address on afterwards.
c; OAMDATA; NES; a; $2004
--Hardware scroll register. Write the x offset and then the y offset, both after reading PPUSTATUS.<br>
--<code>dummy:=PPUSTATUS; PPUSCROLL:=scrollx; PPUSCROLL:=scrolly;</code>
c; PPUSCROLL; NES; a; $2005
--Sets the PPU memory address for PPUDATA. Write the high byte first, then the low byte.<br>
--<code>PPUADDR:=$20; PPUADDR:=$00; // point at the first name table</code>
c; PPUADDR; NES; a; $2006
--Reads or writes PPU memory at the address set through PPUADDR, stepping the address on afterwards. Only safe during vblank.
c; PPUDATA; NES; a; $2007
--Writing a page number here copies that whole 256 byte page into sprite memory in one go. The usual way to update sprites each frame.<br>
--<code>OAMDMA:=2; // copy $0200-$02FF to the sprite table</code>
c; OAMDMA; NES; a; $4014
